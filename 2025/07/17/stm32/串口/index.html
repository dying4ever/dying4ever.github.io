uint8_t rx_buf;

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
	if (huart->Instance ==USART1)//只有这里是 USART1
	{
	HAL_UART_Transmit(huart ,&rx_buf,1,50);
	HAL_UART_Receive_IT(huart,&rx_buf,1);//全是huart
	}
}
