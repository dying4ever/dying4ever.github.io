### msg：

**1.**

​	rosidl_default_generators:用于将自定义的消息文件转化为c++/python源码；

​	builtin_interfaces、geometry_msgs:ros2的消息接口功能包,可以用时间接口Time

```
ros2 pkg create 名字 --build-type ament_cmake --dependencies rosidl_default_generators builtin_interfaces --license Apache-2.0//
```

**2.**

​	新建msg/名字.msg文件（驼峰命名法）

​	如果以来其他包，省略msg，例如：builtin_interface/Time_stamp#记录时间戳

**3.**

​	

CMakeLists.txt

```
cmake_minimum_required(VERSION 3.8)
project(infantry_msgs)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 20)
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# Find packages
find_package(rosidl_default_generators REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)

#************************************************
#使用 file(GLOB ...) 命令来查找当前源代码目录中 msg 文件夹内所有的 .msg 文件，并将这些文件路径存储到 MSG_FILES 变量中。这样，无论你在 msg/ 文件夹中添加多少个 .msg 文件，它们都能自动被包含进来，无需手动列出每个文件。
#RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}：相对路径，相对于当前 CMakeLists.txt 文件所在的目录（CMAKE_CURRENT_SOURCE_DIR）。
#************************************************
file(GLOB MSG_FILES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} "msg/*.msg")
# file(GLOB SRV_FILES RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} "srv/*.srv")
#************************************************
#rosidl_generate_interfaces 用来生成 ROS 2 消息接口文件，它会根据 MSG_FILES 中列出的 .msg 文件自动生成代码（如 C++、Python 等）。
#************************************************
rosidl_generate_interfaces(${PROJECT_NAME}
  ${MSG_FILES}
  # ${SRV_FILES}
  DEPENDENCIES builtin_interfaces geometry_msgs
)
#************************************************
#也可以直接写
#rosidl_generate_interfaces(${PROJECT_NAME}
#  "msg/RobotStatus.msg"
#  "srv/SetRobotTask.srv"
#  "action/NavigateToGoal.action"
# DEPENDENCIES std_msgs action_msgs
#)
#************************************************
ament_package()

```
